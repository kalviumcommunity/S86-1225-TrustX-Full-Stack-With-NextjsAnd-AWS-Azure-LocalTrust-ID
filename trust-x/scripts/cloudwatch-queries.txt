/**
 * CloudWatch Log Queries Reference
 * Sample queries for AWS CloudWatch Logs Insights
 */

// 1. All errors in the last hour
fields @timestamp, level, message, context.requestId, context.endpoint
| filter level = "error"
| sort @timestamp desc
| limit 100

// 2. Errors grouped by endpoint
fields @timestamp, context.endpoint as endpoint
| filter level = "error"
| stats count() by endpoint
| sort count desc

// 3. Slow API requests (>1 second)
fields @timestamp, context.endpoint, context.method, context.duration
| filter context.duration > 1000
| sort context.duration desc
| limit 50

// 4. Average response time by endpoint
fields context.endpoint as endpoint, context.duration as duration
| filter ispresent(duration)
| stats avg(duration) as avg_duration, max(duration) as max_duration, count() as request_count by endpoint
| sort avg_duration desc

// 5. Failed login attempts
fields @timestamp, context.email, context.userId
| filter context.authEvent = "login_failed"
| sort @timestamp desc
| limit 100

// 6. Requests by status code
fields @timestamp, context.statusCode as status
| filter ispresent(status)
| stats count() by status
| sort status asc

// 7. Error rate per 5-minute interval
fields @timestamp
| filter level = "error"
| stats count() as error_count by bin(5m)

// 8. User activity (authenticated requests)
fields @timestamp, context.userId, context.endpoint, context.method
| filter ispresent(context.userId)
| sort @timestamp desc
| limit 100

// 9. Database operation performance
fields @timestamp, context.operation, context.table, context.duration
| filter ispresent(context.table)
| stats avg(context.duration) as avg_duration by context.operation, context.table
| sort avg_duration desc

// 10. Cache hit/miss ratio
fields @timestamp, context.cacheKey
| filter message like /Cache/
| stats count() by message
| sort count desc

// 11. Security events
fields @timestamp, message, context.severity, context.securityEvent
| filter ispresent(context.securityEvent)
| sort @timestamp desc

// 12. Request volume per endpoint
fields context.endpoint as endpoint
| filter ispresent(endpoint)
| stats count() as request_count by endpoint
| sort request_count desc

// 13. Errors by user
fields @timestamp, context.userId, message, context.endpoint
| filter level = "error" and ispresent(context.userId)
| stats count() as error_count by context.userId
| sort error_count desc
| limit 20

// 14. 5xx errors
fields @timestamp, message, context.endpoint, context.statusCode
| filter context.statusCode >= 500
| sort @timestamp desc
| limit 50

// 15. Request correlation (find all logs for a specific request)
fields @timestamp, level, message, context
| filter context.requestId = "YOUR_REQUEST_ID_HERE"
| sort @timestamp asc
